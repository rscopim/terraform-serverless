🐍 Python — Módulo 04

Estruturas de Dados: Listas, Dicts, Tuplas e Sets

Domine as estruturas de dados nativas do Python para manipular coleções de forma eficiente.

1. Listas

listMutável
# Criação e manipulação
frutas = ["maçã", "banana", "uva", "manga"]

# Acesso e slicing
print(frutas[0])       # maçã
print(frutas[-1])      # manga
print(frutas[1:3])     # ['banana', 'uva']
print(frutas[::2])     # ['maçã', 'uva']

# Métodos principais
frutas.append("kiwi")        # adiciona ao final
frutas.insert(1, "laranja")  # insere na posição
frutas.remove("banana")      # remove por valor
frutas.pop()                 # remove último
frutas.sort()                # ordena in-place
frutas.reverse()             # inverte

# Desempacotamento
primeiro, *meio, ultimo = frutas

2. Dicionários

dictChave-Valor
# Criação
servidor = {
    "nome": "web-01",
    "ip": "10.0.1.50",
    "porta": 8080,
    "ativo": True,
    "tags": ["prod", "api"]
}

# Acesso seguro
print(servidor.get("ip"))              # 10.0.1.50
print(servidor.get("regiao", "us-east-1"))  # default

# Métodos úteis
servidor["cpu"] = 4                    # adiciona/atualiza
del servidor["porta"]                  # remove
print(servidor.keys())                 # chaves
print(servidor.values())               # valores
print(servidor.items())                # pares (k, v)

# Iteração
for chave, valor in servidor.items():
    print(f"{chave}: {valor}")

# Merge (Python 3.9+)
config = {"debug": True} | servidor

3. Tuplas e Sets

tupleImutável
# Tuplas são imutáveis
coordenadas = (23.5505, -46.6333)
rgb = (255, 128, 0)

# Desempacotamento
lat, lon = coordenadas

# Named tuple
from collections import namedtuple
Servidor = namedtuple("Servidor", ["nome", "ip", "porta"])
srv = Servidor("web-01", "10.0.1.50", 8080)
print(srv.nome)  # web-01
setÚnicos
# Sets: elementos únicos, sem ordem
ips_prod = {"10.0.1.1", "10.0.1.2", "10.0.1.3"}
ips_dev = {"10.0.1.2", "10.0.1.4"}

# Operações de conjuntos
print(ips_prod | ips_dev)   # união
print(ips_prod & ips_dev)   # interseção
print(ips_prod - ips_dev)   # diferença

# Remover duplicatas
logs = ["erro", "info", "erro", "warn", "info"]
unicos = list(set(logs))
