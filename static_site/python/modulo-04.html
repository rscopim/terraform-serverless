1.2 Indexação e Fatiamento (Slicing)

Acesso por índice e fatiasIndexação
# Índices positivos (da esquerda, começam em 0)
regioes = ["us-east-1", "us-west-2", "eu-west-1", "sa-east-1", "ap-southeast-1"]

print(regioes[0])    # 'us-east-1'  (primeiro)
print(regioes[2])    # 'eu-west-1'  (terceiro)
print(regioes[-1])   # 'ap-southeast-1'  (último)
print(regioes[-2])   # 'sa-east-1'  (penúltimo)

# Slicing: lista[início:fim:passo]
print(regioes[1:3])   # ['us-west-2', 'eu-west-1']  (índice 1 e 2)

print(regioes[2:])    # ['eu-west-1', 'sa-east-1', 'ap-southeast-1']  (do 2 ao fim)
print(regioes[::2])   # ['us-east-1', 'eu-west-1', 'ap-southeast-1']  (passo 2)
print(regioes[::-1])  # lista invertida

# Slicing cria uma NOVA lista (cópia rasa)

1.7 Shallow Copy vs Deep Copy

copy vs deepcopyCópia
import copy

# ═══════════════════════════════════════════════
# CÓPIA RASA (Shallow Copy)
# Copia a lista, mas objetos internos são referências
# ═══════════════════════════════════════════════
original = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]

rasa = original.copy()           # ou: list(original) ou original[:]
rasa[0][0] = 999                 # modifica objeto interno
print(original[0][0])            # 999 — ORIGINAL AFETADO!

rasa.append([10, 11, 12])        # adiciona nova sublista
print(len(original))             # 3 — original não afetado (era sobre a lista principal)

# ═══════════════════════════════════════════════
# CÓPIA PROFUNDA (Deep Copy)
# Copia tudo recursivamente — totalmente independente
# ═══════════════════════════════════════════════
original = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]

profunda = copy.deepcopy(original)
profunda[0][0] = 999
print(original[0][0])            # 1 — original INTACTO!

# ═══════════════════════════════════════════════
# QUANDO USAR CADA UM
# ═══════════════════════════════════════════════
# Shallow copy: lista de imutáveis (int, str, tuple)
ips = ["10.0.1.1", "10.0.1.2", "10.0.1.3"]
backup_ips = ips.copy()  # suficiente! strings são imutáveis

# Deep copy: lista de mutáveis (listas, dicts)
configs = [
    {"nome": "web", "porta": 80, "tags": ["prod"]},
    {"nome": "api", "porta": 8080, "tags": ["prod", "v2"]},
]
backup_configs = copy.deepcopy(configs)  # necessário!

💡 Regra Prática

Use .copy() ou [:] quando a lista contém apenas tipos imutáveis (números, strings, tuplas). Use copy.deepcopy() quando a lista contém objetos mutáveis (outras listas, dicionários, objetos customizados).

💡 Quando NÃO usar comprehension

Evite comprehensions quando: (1) a lógica é complexa demais (mais de 2 condições), (2) você precisa de efeitos colaterais (print, append em outra lista), ou (3) a expressão fica difícil de ler. Nestes casos, use um loop for explícito. Legibilidade sempre vem primeiro!

2.3 Filtragem e Mapeamento

filter(), map() e alternativas pythônicasFuncional
# ═══════════════════════════════════════════════
# filter() — seleciona elementos que passam no teste
# ═══════════════════════════════════════════════
portas = [22, 80, 443, 3306, 5432, 8080, 9090, 27017]

altas = [p for p in portas if p > 1024]

# map() — aplica função a cada elemento
# ═══════════════════════════════════════════════

# Formatar IPs com prefixo
ips = ["192.168.1.1", "192.168.1.2", "192.168.1.3"]

# Pythônico — comprehension:
formatados = [f"host: {ip}" for ip in ips]

# ═══════════════════════════════════════════════
# Combinando filter + map (vs comprehension)
# ═══════════════════════════════════════════════
logs = [

# Merge profundo (deep merge) — não é nativo!
# ═══════════════════════════════════════════════
def deep_merge(base, override):
    """Merge recursivo de dicts aninhados."""
    resultado = base.copy()
    for key, value in override.items():
        if key in resultado and isinstance(resultado[key], dict) and isinstance(value, dict):
            resultado[key] = deep_merge(resultado[key], value)
        else:
            resultado[key] = value
    return resultado

    "cache": {"host": "redis.prod.local"},
}

final = deep_merge(config_base, config_prod)

from collections import OrderedDict, ChainMap

# ═══════════════════════════════════════════════
# OrderedDict — Quando a COMPARAÇÃO de ordem importa
# (Desde Python 3.7, dict normal mantém ordem de inserção,
#  mas OrderedDict considera ordem na comparação ==)
# ═══════════════════════════════════════════════

od1 = OrderedDict([("a", 1), ("b", 2)])

print(od1 == od2)  # False — ordem diferente!

od = OrderedDict([("a", 1), ("b", 2), ("c", 3)])
od.move_to_end("a")        # move 'a' para o final
print(list(od.keys()))     # ['b', 'c', 'a']
od.move_to_end("c", last=False)  # move 'c' para o início
print(list(od.keys()))     # ['c', 'b', 'a']

6.2 Operações de Conjuntos

União, Interseção, Diferença, SimétricaOperações
# Sets para DevOps
portas_web = {80, 443, 8080, 8443}
portas_db = {3306, 5432, 6379, 27017}
portas_abertas = {22, 80, 443, 3306, 8080}
portas_permitidas = {80, 443, 22}

# ═══════════════════════════════════════════════
# UNIÃO (|) — todos os elementos de ambos
# ═══════════════════════════════════════════════
todas_portas = portas_web | portas_db
# {80, 443, 3306, 5432, 6379, 8080, 8443, 27017}

# Equivalente com método:
todas_portas = portas_web.union(portas_db)

# ═══════════════════════════════════════════════
# INTERSEÇÃO (&) — apenas elementos em AMBOS
# ═══════════════════════════════════════════════
portas_web_abertas = portas_web & portas_abertas
# {80, 443, 8080}

# Equivalente:
portas_web_abertas = portas_web.intersection(portas_abertas)

# ═══════════════════════════════════════════════
# DIFERENÇA (-) — elementos no primeiro mas NÃO no segundo
# ═══════════════════════════════════════════════
portas_nao_permitidas = portas_abertas - portas_permitidas
# {3306, 8080} — portas abertas que NÃO deveriam estar!
print(f"⚠️ Portas não autorizadas: {portas_nao_permitidas}")

# Equivalente:
portas_nao_permitidas = portas_abertas.difference(portas_permitidas)

# ═══════════════════════════════════════════════
# DIFERENÇA SIMÉTRICA (^) — em um OU outro, mas não ambos
# ═══════════════════════════════════════════════
apenas_em_um = portas_web ^ portas_abertas
# Portas que estão em web XOR abertas (não nos dois)

# Equivalente:
apenas_em_um = portas_web.symmetric_difference(portas_abertas)

# ═══════════════════════════════════════════════
# SUBCONJUNTO e SUPERCONJUNTO
# ═══════════════════════════════════════════════
portas_https = {443}
print(portas_https <= portas_web)        # True — subset
print(portas_https.issubset(portas_web)) # True

print(portas_web >= portas_https)          # True — superset
print(portas_web.issuperset(portas_https)) # True

# Conjuntos disjuntos (nenhum elemento em comum)
print(portas_web.isdisjoint(portas_db))  # True — sem interseção

# ═══════════════════════════════════════════════
# Operações com múltiplos sets
# ═══════════════════════════════════════════════
set_a = {1, 2, 3, 4}
set_b = {3, 4, 5, 6}
set_c = {4, 5, 6, 7}

# Interseção de todos (elemento que está em TODOS)
comum_todos = set_a & set_b & set_c  # {4}

# União de todos
total = set_a | set_b | set_c  # {1, 2, 3, 4, 5, 6, 7}

💡 Operadores vs Métodos

Os operadores (|, &, -, ^) exigem que ambos os lados sejam sets. Os métodos (.union(), .intersection(), etc.) aceitam qualquer iterável como argumento — mais flexível!

# Operador exige set:
# {1,2,3} | [4,5]  — TypeError!

# Método aceita qualquer iterável:
{1,2,3}.union([4,5])  # {1, 2, 3, 4, 5} ✅

    resultado = ip_teste in ips_bloqueados_lista

inicio = time.perf_counter()
for _ in range(1000):
    resultado = ip_teste in ips_bloqueados_set
tempo_set = time.perf_counter() - inicio

print(f"Lista: {tempo_lista:.4f}s")
print(f"Set:   {tempo_set:.6f}s")
print(f"Set é {tempo_lista/tempo_set:.0f}x mais rápido!")
# Típico: Set é 500-1000x mais rápido para 10K elementos!

ips_permitidos = ["10.0.1.1", "10.0.1.2", "10.0.1.3", ...]

7.4 Fluxograma de Decisão

Algoritmo para escolher a estruturaFluxograma
# ═══════════════════════════════════════════════════════
# FLUXOGRAMA DE DECISÃO:
# ═══════════════════════════════════════════════════════
#
# Precisa mapear CHAVE → VALOR?
#   └─ SIM → DICT ✅
#   └─ NÃO ↓
#
# Elementos devem ser ÚNICOS?
#   └─ SIM → SET ✅  (ou frozenset se precisa ser hashable)
#   └─ NÃO ↓
#
# Dados são FIXOS (não mudam)?
#   └─ SIM → TUPLA ✅  (ou NamedTuple para clareza)
#   └─ NÃO ↓
#
# LISTA ✅
#
# ═══════════════════════════════════════════════════════
# OTIMIZAÇÕES COMUNS:
# ═══════════════════════════════════════════════════════
#
# "Preciso buscar rápido E manter ordem"
#   → Dict (mantém ordem de inserção desde 3.7)
#
# "Preciso de fila (FIFO)"
#   → collections.deque (não lista!)
#
# "Preciso contar elementos"
#   → collections.Counter (subclasse de dict)
#
# "Preciso agrupar por atributo"
#   → collections.defaultdict(list)
#
# "Preciso de prioridade"
#   → heapq (heap sobre lista)

# Verificar SGs com porta aberta para 0.0.0.0/0
def audit_security_groups(infra):
    """Audita security groups por regras permissivas."""
    alertas = []
    for sg_name, sg_config in infra["security_groups"].items():
        for rule in sg_config.get("ingress", []):
            if rule["cidr"] == "0.0.0.0/0" and rule["port"] not in (80, 443):
                alertas.append({
                    "sg": sg_name,
                    "porta": rule["port"],
                    "risco": "Porta não-HTTP aberta para internet"
                })
    return alertas

alertas = audit_security_groups(infrastructure)
for alerta in alertas:
    print(f"⚠️ SG '{alerta['sg']}': porta {alerta['porta']} — {alerta['risco']}")
