Consumo de APIs REST: requests, autenticação, paginação

Integre sistemas consumindo APIs RESTful com a biblioteca requests.

1. Biblioteca requests

requestsHTTP
import requests

# GET simples
response = requests.get("https://api.github.com/users/octocat")
print(response.status_code)  # 200
print(response.json())       # dict com dados

# GET com parâmetros
params = {"q": "python", "sort": "stars", "per_page": 5}
r = requests.get("https://api.github.com/search/repositories", params=params)
repos = r.json()["items"]
for repo in repos:
    print(f"{repo['name']}: ⭐ {repo['stargazers_count']}")

# POST com JSON
payload = {"title": "Bug fix", "body": "Corrigido erro no login"}
r = requests.post(
    "https://api.exemplo.com/issues",
    json=payload,
    headers={"Content-Type": "application/json"}
)

# Timeout e tratamento de erros
try:
    r = requests.get("https://api.exemplo.com/data", timeout=10)
    r.raise_for_status()  # levanta exceção se status >= 400
except requests.exceptions.Timeout:
    print("Timeout na requisição")
except requests.exceptions.HTTPError as e:
    print(f"Erro HTTP: {e.response.status_code}")

2. Autenticação e Headers

AuthSegurança
import os

# Bearer Token
token = os.environ.get("API_TOKEN")
headers = {
    "Authorization": f"Bearer {token}",
    "Accept": "application/json"
}
r = requests.get(
    "https://api.exemplo.com/me",
    headers=headers
)

# Basic Auth
r = requests.get(
    "https://api.exemplo.com/data",
    auth=("usuario", "senha")
)

# Session (reutiliza conexão e headers)
session = requests.Session()
session.headers.update({"Authorization": f"Bearer {token}"})
r1 = session.get("/endpoint1")
r2 = session.get("/endpoint2")
PaginaçãoAPI
# Paginação automática
def buscar_todos(url, headers):
    """Busca todas as páginas de uma API."""
    resultados = []
    page = 1

    while True:
        r = requests.get(
            url,
            headers=headers,
            params={"page": page, "per_page": 100}
        )
        r.raise_for_status()
        dados = r.json()

        if not dados:
            break

        resultados.extend(dados)
        page += 1

    return resultados

# Paginação com next link
def paginar_com_links(url, headers):
    resultados = []
    while url:
        r = requests.get(url, headers=headers)
        resultados.extend(r.json())
        url = r.links.get("next", {}).get("url")
    return resultados
