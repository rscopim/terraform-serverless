Automação AWS com boto3: EC2, S3, Lambda, IAM, CloudWatch

Automatize a infraestrutura AWS usando Python e a SDK oficial boto3.

1. EC2 — Gerenciamento de Instâncias

boto3 EC2AWS
import boto3

ec2 = boto3.resource("ec2", region_name="us-east-1")
client = boto3.client("ec2", region_name="us-east-1")

# Listar instâncias
for instance in ec2.instances.all():
    nome = next((t["Value"] for t in (instance.tags or []) if t["Key"] == "Name"), "N/A")
    print(f"{nome} | {instance.id} | {instance.state['Name']} | {instance.instance_type}")

# Criar instância
instances = ec2.create_instances(
    ImageId="ami-0c02fb55956c7d316",
    InstanceType="t3.micro",
    MinCount=1,
    MaxCount=1,
    KeyName="minha-chave",
    TagSpecifications=[{
        "ResourceType": "instance",
        "Tags": [{"Key": "Name", "Value": "python-demo"}]
    }]
)
print(f"Instância criada: {instances[0].id}")

# Start/Stop
client.stop_instances(InstanceIds=["i-0123456789abcdef0"])
client.start_instances(InstanceIds=["i-0123456789abcdef0"])

2. S3 — Armazenamento de Objetos

boto3 S3AWS
s3 = boto3.client("s3")

# Listar buckets
response = s3.list_buckets()
for bucket in response["Buckets"]:
    print(f"{bucket['Name']} (criado: {bucket['CreationDate']})")

# Upload de arquivo
s3.upload_file("relatorio.pdf", "meu-bucket", "docs/relatorio.pdf")

# Upload com metadados
s3.upload_file(
    "dados.json", "meu-bucket", "data/dados.json",
    ExtraArgs={"ContentType": "application/json", "ServerSideEncryption": "AES256"}
)

# Download
s3.download_file("meu-bucket", "docs/relatorio.pdf", "/tmp/relatorio.pdf")

# Listar objetos
paginator = s3.get_paginator("list_objects_v2")
for page in paginator.paginate(Bucket="meu-bucket", Prefix="logs/"):
    for obj in page.get("Contents", []):
        print(f"{obj['Key']} ({obj['Size']} bytes)")

3. Lambda — Funções Serverless

boto3 LambdaAWS
import json

lambda_client = boto3.client("lambda", region_name="us-east-1")

# Invocar função Lambda
payload = {"nome": "Ricardo", "acao": "processar"}
response = lambda_client.invoke(
    FunctionName="minha-funcao",
    InvocationType="RequestResponse",  # síncrono
    Payload=json.dumps(payload)
)
resultado = json.loads(response["Payload"].read())
print(f"Status: {response['StatusCode']}, Resultado: {resultado}")

# Invocação assíncrona
lambda_client.invoke(
    FunctionName="processar-fila",
    InvocationType="Event",  # assíncrono (fire-and-forget)
    Payload=json.dumps({"batch_id": "abc123"})
)

# Listar funções
paginator = lambda_client.get_paginator("list_functions")
for page in paginator.paginate():
    for func in page["Functions"]:
        print(f"{func['FunctionName']} | Runtime: {func['Runtime']} | Memória: {func['MemorySize']}MB")

4. IAM e CloudWatch

IAMAWS
iam = boto3.client("iam")

# Listar usuários
users = iam.list_users()
for user in users["Users"]:
    print(f"{user['UserName']} | Criado: {user['CreateDate']}")

# Verificar MFA
for user in users["Users"]:
    mfa = iam.list_mfa_devices(UserName=user["UserName"])
    status = "✅" if mfa["MFADevices"] else "❌"
    print(f"{user['UserName']}: MFA {status}")

# Listar políticas do usuário
policies = iam.list_attached_user_policies(
    UserName="ricardo"
)
for p in policies["AttachedPolicies"]:
    print(p["PolicyName"])
CloudWatchAWS
cw = boto3.client("cloudwatch")
logs = boto3.client("logs")

# Métricas de CPU
from datetime import datetime, timedelta
response = cw.get_metric_statistics(
    Namespace="AWS/EC2",
    MetricName="CPUUtilization",
    Dimensions=[{"Name": "InstanceId", "Value": "i-012345"}],
    StartTime=datetime.utcnow() - timedelta(hours=1),
    EndTime=datetime.utcnow(),
    Period=300,
    Statistics=["Average"]
)
for dp in response["Datapoints"]:
    print(f"{dp['Timestamp']}: {dp['Average']:.1f}%")

# Criar alarme
cw.put_metric_alarm(
    AlarmName="CPU-Alta",
    MetricName="CPUUtilization",
    Namespace="AWS/EC2",
    Statistic="Average",
    Period=300,
    EvaluationPeriods=2,
    Threshold=80.0,
    ComparisonOperator="GreaterThanThreshold"
)
