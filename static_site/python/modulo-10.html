1. Invocando Modelos com Bedrock

Bedrock - ClaudeIA
import boto3
import json

bedrock = boto3.client("bedrock-runtime", region_name="us-east-1")

# Invocar Claude (Anthropic)
def invocar_claude(prompt, max_tokens=1024):
    body = json.dumps({
        "anthropic_version": "bedrock-2023-05-31",
        "max_tokens": max_tokens,
        "messages": [
            {"role": "user", "content": prompt}
        ]
    })

    response = bedrock.invoke_model(
        modelId="anthropic.claude-3-sonnet-20240229-v1:0",
        contentType="application/json",
        accept="application/json",
        body=body
    )

    resultado = json.loads(response["body"].read())
    return resultado["content"][0]["text"]

# Uso
resposta = invocar_claude("Explique o que é uma VPC na AWS em 3 frases.")
print(resposta)

# Com contexto de sistema
def invocar_claude_sistema(prompt, sistema="Você é um especialista AWS."):
    body = json.dumps({
        "anthropic_version": "bedrock-2023-05-31",
        "max_tokens": 2048,
        "system": sistema,
        "messages": [{"role": "user", "content": prompt}]
    })
    response = bedrock.invoke_model(
        modelId="anthropic.claude-3-sonnet-20240229-v1:0",
        contentType="application/json",
        accept="application/json",
        body=body
    )
    return json.loads(response["body"].read())["content"][0]["text"]

2. Titan e Embeddings

Titan TextIA
# Invocar Amazon Titan
def invocar_titan(prompt):
    body = json.dumps({
        "inputText": prompt,
        "textGenerationConfig": {
            "maxTokenCount": 512,
            "temperature": 0.7,
            "topP": 0.9
        }
    })

    response = bedrock.invoke_model(
        modelId="amazon.titan-text-express-v1",
        contentType="application/json",
        accept="application/json",
        body=body
    )

    resultado = json.loads(response["body"].read())
    return resultado["results"][0]["outputText"]

resposta = invocar_titan("Liste 5 boas práticas de segurança no S3.")
print(resposta)
EmbeddingsIA
# Gerar embeddings para busca semântica
def gerar_embedding(texto):
    body = json.dumps({
        "inputText": texto
    })

    response = bedrock.invoke_model(
        modelId="amazon.titan-embed-text-v1",
        contentType="application/json",
        accept="application/json",
        body=body
    )

    resultado = json.loads(response["body"].read())
    return resultado["embedding"]  # vetor 1536D

# Similaridade entre textos
import numpy as np

emb1 = gerar_embedding("Como criar um bucket S3?")
emb2 = gerar_embedding("Criação de storage no AWS S3")
emb3 = gerar_embedding("Receita de bolo de chocolate")

def similaridade(a, b):
    return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))

print(f"S3 vs S3: {similaridade(emb1, emb2):.3f}")   # ~0.9
print(f"S3 vs bolo: {similaridade(emb1, emb3):.3f}") # ~0.3

3. Chatbot com Bedrock (Projeto)

chatbot_aws.pyProjeto
"""
Projeto: Chatbot especialista em AWS usando Bedrock
- Mantém histórico de conversa
- Usa Claude como modelo base
- Contexto de sistema como especialista AWS
- Salva conversas em JSON
"""
import boto3
import json
from datetime import datetime

class ChatbotAWS:
    def __init__(self, modelo="anthropic.claude-3-sonnet-20240229-v1:0"):
        self.bedrock = boto3.client("bedrock-runtime", region_name="us-east-1")
        self.modelo = modelo
        self.historico = []
        self.sistema = (
            "Você é um especialista AWS certificado. "
            "Responda de forma clara e prática, com exemplos de código quando relevante. "
            "Sempre mencione boas práticas de segurança e custo."
        )

    def enviar(self, mensagem):
        self.historico.append({"role": "user", "content": mensagem})

        body = json.dumps({
            "anthropic_version": "bedrock-2023-05-31",
            "max_tokens": 2048,
            "system": self.sistema,
            "messages": self.historico
        })

        response = self.bedrock.invoke_model(
            modelId=self.modelo,
            contentType="application/json",
            accept="application/json",
            body=body
        )

        resultado = json.loads(response["body"].read())
        resposta = resultado["content"][0]["text"]
        self.historico.append({"role": "assistant", "content": resposta})
        return resposta

    def salvar_conversa(self):
        arquivo = f"conversa_{datetime.now():%Y%m%d_%H%M%S}.json"
        with open(arquivo, "w", encoding="utf-8") as f:
            json.dump(self.historico, f, indent=2, ensure_ascii=False)
        print(f"💾 Conversa salva em {arquivo}")

if __name__ == "__main__":
    bot = ChatbotAWS()
    print("🤖 Chatbot AWS (digite 'sair' para encerrar)\n")
    while True:
        pergunta = input("Você: ")
        if pergunta.lower() in ("sair", "exit", "quit"):
            bot.salvar_conversa()
            break
        resposta = bot.enviar(pergunta)
        print(f"\n🤖 {resposta}\n")
