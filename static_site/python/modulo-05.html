2. CSV e JSON

csvDados
import csv

# Leitura CSV
with open("servidores.csv") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(row["nome"], row["ip"])

# Escrita CSV
dados = [
    {"nome": "web-01", "ip": "10.0.1.1"},
    {"nome": "web-02", "ip": "10.0.1.2"},
]
with open("output.csv", "w", newline="") as f:
    writer = csv.DictWriter(f, fieldnames=["nome", "ip"])
    writer.writeheader()
    writer.writerows(dados)

3. Tratamento de Exceções

try / exceptExceções
# Tratamento completo
try:
    with open("dados.json") as f:
        dados = json.load(f)
    resultado = dados["valor"] / dados["divisor"]
except FileNotFoundError:
    print("Arquivo não encontrado!")
    dados = {}
except KeyError as e:
    print(f"Chave ausente: {e}")
except ZeroDivisionError:
    print("Divisão por zero!")
except Exception as e:
    print(f"Erro inesperado: {type(e).__name__}: {e}")
else:
    print(f"Sucesso! Resultado: {resultado}")
finally:
    print("Bloco finally sempre executa")

# Levantar exceções
def validar_idade(idade):
    if idade < 0 or idade > 150:
        raise ValueError(f"Idade inválida: {idade}")
    return idade
