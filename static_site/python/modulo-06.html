Orientação a Objetos: Classes, Herança e Polimorfismo

OOP permite modelar sistemas complexos com classes, encapsulamento, herança e polimorfismo.

1. Classes e Objetos

classOOP
class Servidor:
    """Representa um servidor na infraestrutura."""

    total_servidores = 0  # atributo de classe

    def __init__(self, nome, ip, porta=8080):
        self.nome = nome          # público
        self.ip = ip
        self._porta = porta       # protegido (convenção)
        self.__status = "stopped" # privado (name mangling)
        Servidor.total_servidores += 1

    @property
    def porta(self):
        """Getter com @property."""
        return self._porta

    @porta.setter
    def porta(self, valor):
        """Setter com validação."""
        if not 1 <= valor <= 65535:
            raise ValueError("Porta inválida")
        self._porta = valor

    def iniciar(self):
        self.__status = "running"
        return f"{self.nome} iniciado em {self.ip}:{self._porta}"

    def __str__(self):
        return f"Servidor({self.nome}, {self.ip})"

    def __repr__(self):
        return f"Servidor('{self.nome}', '{self.ip}', {self._porta})"

# Uso
srv = Servidor("web-01", "10.0.1.50")
srv.porta = 443
print(srv.iniciar())

2. Herança e Polimorfismo

HerançaOOP
from abc import ABC, abstractmethod

class RecursoAWS(ABC):
    """Classe abstrata base."""

    def __init__(self, nome, regiao="us-east-1"):
        self.nome = nome
        self.regiao = regiao

    @abstractmethod
    def criar(self):
        pass

    @abstractmethod
    def deletar(self):
        pass

class InstanciaEC2(RecursoAWS):
    def __init__(self, nome, tipo="t3.micro", **kwargs):
        super().__init__(nome, **kwargs)
        self.tipo = tipo

    def criar(self):
        return f"EC2 '{self.nome}' ({self.tipo}) criada em {self.regiao}"

    def deletar(self):
        return f"EC2 '{self.nome}' terminada"

class BucketS3(RecursoAWS):
    def criar(self):
        return f"Bucket '{self.nome}' criado em {self.regiao}"

    def deletar(self):
        return f"Bucket '{self.nome}' deletado"

# Polimorfismo
recursos = [
    InstanciaEC2("web-server", tipo="t3.small"),
    BucketS3("meu-bucket", regiao="sa-east-1"),
]
for recurso in recursos:
    print(recurso.criar())  # cada um executa seu método
