classOOP
class Servidor:
    """Representa um servidor na infraestrutura."""

    total_servidores = 0  # atributo de classe

    def __init__(self, nome, ip, porta=8080):
        self.nome = nome          # público
        self.ip = ip
        self._porta = porta       # protegido (convenção)
        self.__status = "stopped" # privado (name mangling)
        Servidor.total_servidores += 1

    @property
    def porta(self):
        """Getter com @property."""
        return self._porta

    @porta.setter
    def porta(self, valor):
        """Setter com validação."""
        if not 1 <= valor <= 65535:
            raise ValueError("Porta inválida")
        self._porta = valor

    def iniciar(self):
        self.__status = "running"
        return f"{self.nome} iniciado em {self.ip}:{self._porta}"

    def __str__(self):
        return f"Servidor({self.nome}, {self.ip})"

    def __repr__(self):
        return f"Servidor('{self.nome}', '{self.ip}', {self._porta})"

# Uso
srv = Servidor("web-01", "10.0.1.50")
srv.porta = 443
print(srv.iniciar())

2. Herança e Polimorfismo

HerançaOOP
from abc import ABC, abstractmethod

class RecursoAWS(ABC):
    """Classe abstrata base."""

    def __init__(self, nome, regiao="us-east-1"):
        self.nome = nome
        self.regiao = regiao

    @abstractmethod
    def criar(self):
        pass

    @abstractmethod
    def deletar(self):
        pass

class InstanciaEC2(RecursoAWS):
    def __init__(self, nome, tipo="t3.micro", **kwargs):
        super().__init__(nome, **kwargs)
        self.tipo = tipo

    def criar(self):
        return f"EC2 '{self.nome}' ({self.tipo}) criada em {self.regiao}"

    def deletar(self):
        return f"EC2 '{self.nome}' terminada"

class BucketS3(RecursoAWS):
    def criar(self):
        return f"Bucket '{self.nome}' criado em {self.regiao}"

    def deletar(self):
        return f"Bucket '{self.nome}' deletado"

# Polimorfismo
recursos = [
    InstanciaEC2("web-server", tipo="t3.small"),
    BucketS3("meu-bucket", regiao="sa-east-1"),
]
for recurso in recursos:
    print(recurso.criar())  # cada um executa seu método

def somar(a, b):
    return a + b

PaginaçãoAPI
# Paginação automática
def buscar_todos(url, headers):
    """Busca todas as páginas de uma API."""
    resultados = []
    page = 1

    while True:
        r = requests.get(
            url,
            headers=headers,
            params={"page": page, "per_page": 100}
        )
        r.raise_for_status()
        dados = r.json()

        if not dados:
            break

        resultados.extend(dados)
        page += 1

    return resultados

# Paginação com next link
def paginar_com_links(url, headers):
    resultados = []
    while url:
        r = requests.get(url, headers=headers)
        resultados.extend(r.json())
        url = r.links.get("next", {}).get("url")
    return resultados

def invocar_claude(prompt, max_tokens=1024):
    body = json.dumps({
        "anthropic_version": "bedrock-2023-05-31",
        "max_tokens": max_tokens,
        "messages": [
            {"role": "user", "content": prompt}
        ]
    })

    response = bedrock.invoke_model(
        modelId="anthropic.claude-3-sonnet-20240229-v1:0",
        contentType="application/json",
        accept="application/json",
        body=body
    )

    resultado = json.loads(response["body"].read())
    return resultado["content"][0]["text"]

"""
Projeto: Script que coleta inventário completo da conta AWS
- Lista EC2, S3, Lambda, RDS
- Gera relatório CSV e JSON
- Envia alerta se encontrar recursos sem tags
"""
import boto3
import csv
import json
from datetime import datetime

def coletar_ec2(regiao):
    ec2 = boto3.client("ec2", region_name=regiao)
    instances = []
    paginator = ec2.get_paginator("describe_instances")

    for page in paginator.paginate():
        for reservation in page["Reservations"]:
            for inst in reservation["Instances"]:
                tags = {t["Key"]: t["Value"] for t in inst.get("Tags", [])}
                instances.append({
                    "id": inst["InstanceId"],
                    "tipo": inst["InstanceType"],
                    "estado": inst["State"]["Name"],
                    "nome": tags.get("Name", "SEM-NOME"),
                    "regiao": regiao
                })
    return instances

def gerar_relatorio(dados, formato="csv"):
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    if formato == "csv":
        with open(f"inventario_{timestamp}.csv", "w", newline="") as f:
            writer = csv.DictWriter(f, fieldnames=dados[0].keys())
            writer.writeheader()
            writer.writerows(dados)
    else:
        with open(f"inventario_{timestamp}.json", "w") as f:
            json.dump(dados, f, indent=2, default=str)

if __name__ == "__main__":
    regioes = ["us-east-1", "sa-east-1"]
    todas_instancias = []
    for regiao in regioes:
        todas_instancias.extend(coletar_ec2(regiao))
    gerar_relatorio(todas_instancias)
    print(f"✅ {len(todas_instancias)} instâncias catalogadas")

Projeto 2: CLI de Gerenciamento de Servidores

server_cli.pyProjeto
"""
Projeto: CLI interativa para gerenciar instâncias EC2
- Start/Stop/Terminate instâncias
- Filtrar por tags, estado, tipo
- Exibir custos estimados
"""
import argparse
import boto3

def listar_instancias(filtro_estado=None):
    ec2 = boto3.resource("ec2")
    filters = []
    if filtro_estado:
        filters.append({"Name": "instance-state-name", "Values": [filtro_estado]})

    for inst in ec2.instances.filter(Filters=filters):
        tags = {t["Key"]: t["Value"] for t in (inst.tags or [])}
        print(f"  {tags.get('Name', 'N/A'):20} | {inst.id} | {inst.state['Name']:10} | {inst.instance_type}")

def gerenciar(instance_ids, acao):
    ec2 = boto3.client("ec2")
    acoes = {
        "start": ec2.start_instances,
        "stop": ec2.stop_instances,
        "terminate": ec2.terminate_instances,
    }
    if acao in acoes:
        acoes[acao](InstanceIds=instance_ids)
        print(f"✅ {acao} executado em {instance_ids}")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Gerenciador EC2")
    parser.add_argument("acao", choices=["listar", "start", "stop", "terminate"])
    parser.add_argument("--ids", nargs="+", help="IDs das instâncias")
    parser.add_argument("--estado", help="Filtrar por estado")
    args = parser.parse_args()

    if args.acao == "listar":
        listar_instancias(args.estado)
    else:
        gerenciar(args.ids, args.acao)

Projeto 3: Chatbot AWS com Bedrock

chatbot_aws.pyProjeto
"""
Projeto: Chatbot especialista em AWS usando Bedrock
- Mantém histórico de conversa
- Usa Claude como modelo base
- Contexto de sistema como especialista AWS
- Salva conversas em JSON
"""
import boto3
import json
from datetime import datetime

class ChatbotAWS:
    def __init__(self, modelo="anthropic.claude-3-sonnet-20240229-v1:0"):
        self.bedrock = boto3.client("bedrock-runtime", region_name="us-east-1")
        self.modelo = modelo
        self.historico = []
        self.sistema = (
            "Você é um especialista AWS certificado. "
            "Responda de forma clara e prática, com exemplos de código quando relevante. "
            "Sempre mencione boas práticas de segurança e custo."
        )

    def enviar(self, mensagem):
        self.historico.append({"role": "user", "content": mensagem})

        body = json.dumps({
            "anthropic_version": "bedrock-2023-05-31",
            "max_tokens": 2048,
            "system": self.sistema,
            "messages": self.historico
        })

        response = self.bedrock.invoke_model(
            modelId=self.modelo,
            contentType="application/json",
            accept="application/json",
            body=body
        )

        resultado = json.loads(response["body"].read())
        resposta = resultado["content"][0]["text"]
        self.historico.append({"role": "assistant", "content": resposta})
        return resposta

    def salvar_conversa(self):
        arquivo = f"conversa_{datetime.now():%Y%m%d_%H%M%S}.json"
        with open(arquivo, "w", encoding="utf-8") as f:
            json.dump(self.historico, f, indent=2, ensure_ascii=False)
        print(f"💾 Conversa salva em {arquivo}")

if __name__ == "__main__":
    bot = ChatbotAWS()
    print("🤖 Chatbot AWS (digite 'sair' para encerrar)\n")
    while True:
        pergunta = input("Você: ")
        if pergunta.lower() in ("sair", "exit", "quit"):
            bot.salvar_conversa()
            break
        resposta = bot.enviar(pergunta)
        print(f"\n🤖 {resposta}\n")
