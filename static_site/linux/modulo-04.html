Rede

Networking Básico — Diagnóstico e Inspeção

Os comandos de rede são essenciais para troubleshooting de conectividade, diagnóstico de problemas DNS e análise de tráfego em servidores Linux.

ip — Ferramenta moderna de rede

O comando ip substituiu o antigo ifconfig. É a ferramenta padrão para gerenciar interfaces, endereços e rotas no Linux moderno.

ip addr — Endereços IPEndereços
# Listar todas as interfaces e IPs:
ip addr show
ip a                      # forma abreviada

# Mostrar apenas interface específica:
ip addr show eth0
ip a show dev ens33

# Informações resumidas:
ip -br addr               # brief format (limpo)
# lo      UNKNOWN  127.0.0.1/8
# eth0    UP       192.168.1.100/24

# Adicionar IP manualmente (temporário):
sudo ip addr add 192.168.1.200/24 dev eth0

# Remover IP:
sudo ip addr del 192.168.1.200/24 dev eth0

# Ativar/desativar interface:
sudo ip link set eth0 up
sudo ip link set eth0 down

# Ver apenas IPv4:
ip -4 addr show

# Ver apenas IPv6:
ip -6 addr show

# Informações da interface (MAC, MTU, estado):
ip link show eth0
ip route — Tabela de rotasRotas
# Ver tabela de rotas:
ip route show
ip r                      # forma abreviada
# default via 192.168.1.1 dev eth0
# 192.168.1.0/24 dev eth0 proto kernel

# Qual rota será usada para um destino?
ip route get 8.8.8.8
# 8.8.8.8 via 192.168.1.1 dev eth0

# Adicionar rota estática:
sudo ip route add 10.0.0.0/8 via 192.168.1.1

# Adicionar rota default (gateway):
sudo ip route add default via 192.168.1.1

# Remover rota:
sudo ip route del 10.0.0.0/8

# Ver tabela ARP (vizinhos):
ip neigh show
ip n

# Limpar cache ARP:
sudo ip neigh flush all

# Ver estatísticas da interface:
ip -s link show eth0

ss — Socket Statistics (substituto do netstat)

ss — Conexões e portasSockets
# Portas TCP em escuta:
ss -tlnp
# -t = TCP
# -l = listening (em escuta)
# -n = numérico (sem resolver DNS)
# -p = processo associado

# Portas UDP em escuta:

# Todas as conexões TCP estabelecidas:

# Conexões em um estado específico:

ss state close-wait
ss state syn-sent

# Filtrar por porta:
ss -tlnp | grep :80
ss -tlnp sport = :443

# Filtrar por IP destino:
ss -tnp dst 10.0.0.1

# Contar conexões por estado:
ss -s
# Total: 285
# TCP: 42 (estab 15, closed 2, ...)

# Quem está conectado na porta 22 (SSH)?
ss -tnp sport = :22

# Conexões do nginx:
ss -tnp | grep nginx

Diagnóstico de conectividade

ping — Teste de conectividadeICMP
# Ping básico (Ctrl+C para parar):
ping google.com

# Número limitado de pacotes (-c):
ping -c 4 192.168.1.1

# Intervalo personalizado (-i):
ping -i 0.5 192.168.1.1    # a cada 0.5s

# Timeout por pacote (-W):
ping -W 2 -c 3 10.0.0.1    # timeout 2s

# Tamanho do pacote (-s):
ping -s 1472 192.168.1.1   # testa MTU

# Ping com timestamp:
ping -c 5 -D google.com

# Verificar se host está vivo (silencioso):
ping -c 1 -W 1 host &>/dev/null && echo "UP" || echo "DOWN"

# Ping em rede local (descobrir hosts):
for i in {1..254}; do
  ping -c 1 -W 1 192.168.1.$i &>/dev/null && \
    echo "192.168.1.$i está ativo"
done
traceroute — Rota dos pacotesTraceroute
# Traceroute padrão (UDP):
traceroute google.com
# 1  gateway (192.168.1.1)    1.234 ms
# 2  isp-router (10.0.0.1)   5.678 ms
# 3  * * *                    (filtrado)
# 4  google-edge              12.345 ms

# Traceroute com ICMP (-I):
sudo traceroute -I google.com

# Traceroute TCP (-T, porta específica):
sudo traceroute -T -p 443 google.com

# Número máximo de hops (-m):
traceroute -m 15 destino.com

# Alternativa moderna — mtr (combina ping + traceroute):
mtr google.com              # interativo
mtr -r -c 10 google.com    # relatório (10 ciclos)

# mtr mostra perda de pacotes por hop:
# HOST                Loss%  Snt  Last  Avg  Best  Wrst
# 1. gateway          0.0%   10   1.2   1.1  0.9   1.5
# 2. isp              5.0%   10   5.3   5.8  4.2   8.1
# 3. destino          0.0%   10  12.1  11.8 10.5  14.2

# Instalar mtr:
sudo apt install mtr-tiny

DNS — dig e nslookup

dig — DNS lookup detalhadoDNS
# Consulta DNS básica:
dig google.com

# Apenas a resposta (+short):
dig +short google.com
# 142.250.79.46

# Tipo de registro específico:
dig google.com A          # IPv4
dig google.com AAAA       # IPv6
dig google.com MX         # Mail servers
dig google.com NS         # Name servers
dig google.com TXT        # TXT records
dig google.com CNAME      # Aliases
dig google.com SOA        # Start of Authority

# Usar DNS específico (@):
dig @8.8.8.8 meusite.com
dig @1.1.1.1 meusite.com

# Consulta reversa (IP → nome):
dig -x 8.8.8.8

# Trace completo (resolver desde root):
dig +trace google.com

# Verificar propagação DNS:
dig @8.8.8.8 +short meudominio.com
dig @1.1.1.1 +short meudominio.com
dig @208.67.222.222 +short meudominio.com

# Ver TTL:
dig +noall +answer google.com

# Todas as informações:
dig +noall +answer +authority +additional google.com ANY
nslookup — Consulta DNS simplesnslookup
# Consulta básica:
nslookup google.com
# Server:   127.0.0.53
# Address:  127.0.0.53#53
# Name:     google.com
# Address:  142.250.79.46

# Usando servidor DNS específico:
nslookup google.com 8.8.8.8

# Tipo de registro:
nslookup -type=MX google.com
nslookup -type=NS google.com
nslookup -type=TXT google.com

# Consulta reversa:
nslookup 8.8.8.8

# host — alternativa mais simples:
host google.com
host -t MX google.com
host -t NS google.com
host 8.8.8.8              # reversa

# Verificar resolução local:
getent hosts google.com
# Usa a mesma cadeia de resolução do sistema
# (inclui /etc/hosts + DNS)

# Limpar cache DNS local:
sudo systemd-resolve --flush-caches
# ou
sudo resolvectl flush-caches

# Ver status do resolver:
resolvectl status
🔍 Fluxo de resolução DNS no Linux: 
Aplicação → /etc/nsswitch.conf → /etc/hosts → systemd-resolved → /etc/resolv.conf → DNS Server
                                      ↓
                              Ordem de busca:
                              1. /etc/hosts (local, sem rede)
                              2. Cache do systemd-resolved
                              3. DNS configurado em /etc/resolv.conf
                              4. DNS do DHCP ou manual
Configuração

Configuração de Rede

Configurar interfaces de rede, IP estático, DHCP e DNS é fundamental para administração de servidores. No Ubuntu moderno, usamos Netplan como camada de configuração.

Netplan — Configuração moderna (Ubuntu 18.04+)

Netplan é o sistema de configuração de rede padrão do Ubuntu. Usa arquivos YAML em /etc/netplan/ e gera configurações para NetworkManager ou systemd-networkd.

IP Estático com NetplanStatic IP
# Arquivo: /etc/netplan/01-netcfg.yaml
# ou: /etc/netplan/00-installer-config.yaml

network:
  version: 2
  renderer: networkd      # ou NetworkManager
  ethernets:
    ens33:                 # nome da interface
      dhcp4: no
      addresses:
        - 192.168.1.100/24
      routes:
        - to: default
          via: 192.168.1.1
      nameservers:
        addresses:
          - 8.8.8.8
          - 8.8.4.4
        search:
          - meudominio.local

# Aplicar configuração:
sudo netplan apply

# Testar antes de aplicar (reverte em 120s):
sudo netplan try

# Ver configuração gerada:
sudo netplan get

# Validar YAML:
sudo netplan generate
DHCP com NetplanDHCP
# Configuração DHCP (padrão):
network:
  version: 2
  renderer: networkd
  ethernets:
    ens33:
      dhcp4: yes
      dhcp6: no

# DHCP com DNS manual override:
network:
  version: 2
  ethernets:
    ens33:
      dhcp4: yes
      dhcp4-overrides:
        use-dns: no       # ignora DNS do DHCP
      nameservers:
        addresses:
          - 1.1.1.1
          - 1.0.0.1

# Múltiplas interfaces:
network:
  version: 2
  ethernets:
    ens33:
      dhcp4: yes
    ens34:
      dhcp4: no
      addresses:
        - 10.0.0.50/24

# WiFi com Netplan:
network:
  version: 2
  wifis:
    wlp2s0:
      dhcp4: yes
      access-points:
        "MinhaRede":
          password: "senhasegura123"

Interfaces de rede — Conceitos

Nomenclatura e conceitosInterfaces
# Nomenclatura moderna (Predictable Network Interface Names):
# ens33   = Ethernet, slot PCI 33
# enp0s3  = Ethernet, bus PCI 0, slot 3
# eno1    = Ethernet onboard #1
# wlp2s0  = Wireless, bus PCI 2, slot 0
# lo      = Loopback (127.0.0.1)
# docker0 = Bridge do Docker
# br-xxx  = Bridge criada manualmente
# veth... = Virtual ethernet (containers)

# Nomenclatura legada (desabilitada por padrão):
# eth0, eth1, wlan0 (não recomendado)

# Ver todas as interfaces:
ip link show
# 1: lo: <LOOPBACK,UP> mtu 65536
# 2: ens33: <BROADCAST,MULTICAST,UP> mtu 1500
# 3: docker0: <NO-CARRIER,BROADCAST> mtu 1500

# Estados da interface:
# UP        = ativa e funcionando
# DOWN      = desativada
# NO-CARRIER = sem cabo/sem conexão física
# LOWER_UP  = camada física conectada

# Ver velocidade e duplex:
ethtool ens33
# Speed: 1000Mb/s
# Duplex: Full
# Link detected: yes

# Instalar ethtool:
sudo apt install ethtool
Firewall

Firewall — iptables e UFW

O firewall é a primeira linha de defesa de um servidor. No Linux, iptables é o motor por trás do firewall do kernel, e UFW (Uncomplicated Firewall) é a interface simplificada para o dia a dia.

iptables — Fundamentos

iptables opera com chains (INPUT, OUTPUT, FORWARD) e policies (ACCEPT, DROP, REJECT). Cada pacote de rede passa pelas regras na ordem até encontrar um match.

iptables — Conceitos e visualizaçãoiptables
# Estrutura: tabelas → chains → regras
# Tabela filter (padrão):
#   INPUT    = pacotes destinados ao servidor
#   OUTPUT   = pacotes saindo do servidor
#   FORWARD  = pacotes passando pelo servidor (router)

# Ver regras atuais:
sudo iptables -L -n -v
# -L = list, -n = numérico, -v = verbose

# Ver com números de regra:
sudo iptables -L --line-numbers

# Ver regras em formato para script:
sudo iptables -S

# Política padrão (o que acontece se nenhuma regra bater):
sudo iptables -P INPUT DROP      # negar tudo por padrão
sudo iptables -P OUTPUT ACCEPT   # permitir saída
sudo iptables -P FORWARD DROP    # negar forward

# ⚠️ CUIDADO: -P INPUT DROP pode te trancar fora!
# Sempre adicione regra SSH ANTES de mudar policy!
iptables — Regras comunsRegras
# Permitir loopback:
sudo iptables -A INPUT -i lo -j ACCEPT

# Permitir conexões já estabelecidas:
sudo iptables -A INPUT -m state \
  --state ESTABLISHED,RELATED -j ACCEPT

# Permitir SSH (porta 22):
sudo iptables -A INPUT -p tcp --dport 22 -j ACCEPT

# Permitir HTTP e HTTPS:
sudo iptables -A INPUT -p tcp --dport 80 -j ACCEPT
sudo iptables -A INPUT -p tcp --dport 443 -j ACCEPT

# Permitir ping (ICMP):
sudo iptables -A INPUT -p icmp --icmp-type echo-request -j ACCEPT

# Bloquear IP específico:
sudo iptables -A INPUT -s 192.168.1.50 -j DROP

# Permitir range de IPs:
sudo iptables -A INPUT -s 10.0.0.0/24 -p tcp --dport 3306 -j ACCEPT

# Limitar rate (anti brute-force SSH):
sudo iptables -A INPUT -p tcp --dport 22 \
  -m connlimit --connlimit-above 3 -j REJECT

# Deletar regra por número:
sudo iptables -D INPUT 5

# Limpar TODAS as regras:
sudo iptables -F

# Salvar regras (persistir):
sudo apt install iptables-persistent
sudo netfilter-persistent save
⚡ Ordem das regras importa! iptables processa regras de cima para baixo. A primeira regra que faz match é executada. Coloque regras mais específicas primeiro e as genéricas (como DROP) por último.

UFW — Uncomplicated Firewall

UFW é a camada de abstração sobre iptables recomendada para Ubuntu. Muito mais simples de usar no dia a dia, mantendo a mesma segurança.

UFW — Comandos essenciaisUFW Básico
# Instalar (geralmente já vem):
sudo apt install ufw

# Ver status:
sudo ufw status
sudo ufw status verbose
sudo ufw status numbered       # com números

# ⚠️ ANTES de ativar, permita SSH:
sudo ufw allow ssh
# ou
sudo ufw allow 22/tcp

# Ativar firewall:
sudo ufw enable

# Desativar:
sudo ufw disable

# Política padrão (recomendado):
sudo ufw default deny incoming
sudo ufw default allow outgoing

# Permitir portas:
sudo ufw allow 80/tcp          # HTTP
sudo ufw allow 443/tcp         # HTTPS
sudo ufw allow 3000            # TCP + UDP na porta 3000
sudo ufw allow 8080:8090/tcp   # Range de portas

# Permitir por nome de serviço:
sudo ufw allow 'Nginx Full'    # HTTP + HTTPS
sudo ufw allow 'OpenSSH'
sudo ufw app list              # ver perfis disponíveis
UFW — Regras avançadasUFW Avançado
# Permitir de IP específico:
sudo ufw allow from 192.168.1.0/24

# Permitir IP em porta específica:
sudo ufw allow from 10.0.0.50 to any port 3306

# Negar (bloquear):
sudo ufw deny from 203.0.113.50
sudo ufw deny in on eth0 to any port 25

# Rejeitar (envia resposta ao cliente):
sudo ufw reject 23/tcp

# Deletar regra:
sudo ufw status numbered
sudo ufw delete 3              # por número
sudo ufw delete allow 8080/tcp # por regra

# Limitar conexões (anti brute-force):
sudo ufw limit ssh
# Bloqueia IPs com mais de 6 conexões em 30s

# Logging:
sudo ufw logging on
sudo ufw logging medium        # low/medium/high/full
# Logs em /var/log/ufw.log

# Reset (remover todas as regras):
sudo ufw reset

# Regras para IPv6:
# Editar /etc/default/ufw → IPV6=yes
sudo ufw reload
💡 Setup de firewall para servidor web: 
# Script típico para servidor web:
sudo ufw default deny incoming
sudo ufw default allow outgoing
sudo ufw allow ssh               # SEMPRE primeiro!
sudo ufw allow 'Nginx Full'     # HTTP + HTTPS
sudo ufw limit ssh               # anti brute-force
sudo ufw enable
sudo ufw status verbose
SSH

SSH — Acesso Remoto Seguro

SSH (Secure Shell) é o protocolo padrão para administração remota de servidores Linux. Permite login seguro, transferência de arquivos e tunelamento de conexões — tudo criptografado.

Conexão e autenticação

ssh — Conectar a servidoresConexão
# Conexão básica:
ssh usuario@192.168.1.100
ssh root@meuservidor.com

# Porta personalizada (-p):
ssh -p 2222 admin@server.com

# Verbose (debug de conexão):
ssh -v usuario@server.com    # 1 nível
ssh -vv usuario@server.com   # 2 níveis (mais detalhes)

# Executar comando remoto (sem shell interativo):
ssh user@server "uptime"
ssh user@server "df -h && free -m"

# Executar script local no servidor remoto:
ssh user@server 'bash -s' < script_local.sh

# Manter sessão viva (evitar timeout):
ssh -o ServerAliveInterval=60 user@server

# Usar chave privada específica (-i):
ssh -i ~/.ssh/minha-chave.pem ec2-user@aws-instance

# SSH sem verificar host key (lab apenas!):
ssh -o StrictHostKeyChecking=no user@server

# Escape sequences (dentro de sessão SSH):
# ~.  = desconectar forçado
# ~?  = listar escapes
# ~~  = enviar ~ literal
ssh-keygen — Gerar chavesChaves
# Gerar par de chaves (Ed25519 - recomendado):
ssh-keygen -t ed25519 -C "email@exemplo.com"
# Gera: ~/.ssh/id_ed25519 (privada)
#        ~/.ssh/id_ed25519.pub (pública)

# Gerar RSA 4096 bits:
ssh-keygen -t rsa -b 4096 -C "email@exemplo.com"

# Gerar com nome personalizado:
ssh-keygen -t ed25519 -f ~/.ssh/servidor-prod

# Sem passphrase (automação):
ssh-keygen -t ed25519 -f ~/.ssh/deploy-key -N ""

# Copiar chave pública para o servidor:
ssh-copy-id usuario@192.168.1.100
ssh-copy-id -i ~/.ssh/id_ed25519.pub user@server
ssh-copy-id -p 2222 user@server

# O que ssh-copy-id faz:
# Adiciona sua chave pública ao arquivo
# ~/.ssh/authorized_keys do servidor

# Verificar fingerprint da chave:
ssh-keygen -l -f ~/.ssh/id_ed25519.pub
# 256 SHA256:abc123... email@exemplo.com (ED25519)

# Mudar passphrase de chave existente:
ssh-keygen -p -f ~/.ssh/id_ed25519

# Permissões corretas (obrigatório!):
chmod 700 ~/.ssh
chmod 600 ~/.ssh/id_ed25519        # privada
chmod 644 ~/.ssh/id_ed25519.pub    # pública
chmod 600 ~/.ssh/authorized_keys

SSH Config — Atalhos para conexões

~/.ssh/config — Configuração de hostsConfig
# Arquivo: ~/.ssh/config
# Permite criar "atalhos" para conexões frequentes

# Servidor de produção:
Host prod
    HostName 54.123.45.67
    User ubuntu
    Port 22
    IdentityFile ~/.ssh/aws-prod.pem
    ServerAliveInterval 60

# Servidor de desenvolvimento:
Host dev
    HostName 192.168.1.100
    User devops
    Port 2222
    IdentityFile ~/.ssh/id_ed25519

# Jump host (bastion):
Host bastion
    HostName 52.10.20.30
    User admin
    IdentityFile ~/.ssh/bastion.pem

Host interno-via-bastion
    HostName 10.0.1.50
    User app
    ProxyJump bastion

# Wildcard para todos os servidores AWS:
Host *.aws
    User ec2-user
    IdentityFile ~/.ssh/aws-key.pem
    StrictHostKeyChecking no

# Configuração global:
Host *
    ServerAliveInterval 60
    ServerAliveCountMax 3
    AddKeysToAgent yes

# USO: agora basta digitar:
ssh prod            # em vez de: ssh -i ~/.ssh/aws-prod.pem ubuntu@54.123.45.67
ssh dev             # em vez de: ssh -p 2222 devops@192.168.1.100

SSH Tunneling — Port Forwarding

-L — Local Port ForwardLocal Tunnel
# Cenário: acessar serviço remoto via porta local
# Sintaxe: ssh -L local_port:destino:remote_port user@server

# Acessar banco remoto na porta local 5432:
ssh -L 5432:localhost:5432 user@db-server
# Agora: psql -h localhost -p 5432

# Acessar painel admin que só aceita localhost:
ssh -L 8080:localhost:9090 user@server
# Agora abra: http://localhost:8080

# Acessar host interno via jump:
ssh -L 3306:10.0.1.50:3306 user@bastion
# Acessa MySQL do host interno via bastion

# Rodar em background (-fN = background, sem shell):
ssh -fNL 5432:localhost:5432 user@db-server

# Múltiplos tunnels na mesma conexão:
ssh -L 5432:localhost:5432 \
    -L 6379:localhost:6379 \
    user@server
-R — Remote Port ForwardRemote Tunnel
# Cenário: expor serviço local para o servidor remoto
# Sintaxe: ssh -R remote_port:localhost:local_port user@server

# Expor app local (porta 3000) na porta 8080 do servidor:
ssh -R 8080:localhost:3000 user@server
# Agora no servidor: curl http://localhost:8080

# Útil para:
# - Mostrar app local para colega remoto
# - Webhook testing (expor localhost)
# - Bypass de NAT

# Dynamic port forwarding (SOCKS proxy):
ssh -D 9090 user@server
# Configura browser para proxy SOCKS5 localhost:9090
# Todo tráfego passa pelo servidor (VPN simples)

# Manter tunnel ativo com autossh:
sudo apt install autossh
autossh -M 0 -fNL 5432:localhost:5432 user@server
# Reconecta automaticamente se cair

Transferência de arquivos — scp e rsync

scp — Cópia segurascp
# Local → Remoto:
scp arquivo.txt user@server:/home/user/

# Remoto → Local:

scp user@srv1:/dados/file user@srv2:/backup/

# Porta personalizada:
scp -P 2222 arquivo.txt user@server:/tmp/

# Com chave específica:
scp -i ~/.ssh/key.pem file user@server:/path/

# Preservar atributos:
scp -p arquivo.txt user@server:/path/
rsync — Sincronização inteligentersync
# rsync > scp: transfere apenas diferenças!

# Sync local → remoto:
rsync -avz /var/www/ user@server:/var/www/
# -a = archive (preserva tudo)
# -v = verbose
# -z = compressão durante transferência

# Sync remoto → local:
rsync -avz user@server:/var/log/ ./logs/

# Com progresso:
rsync -avz --progress origem/ destino/

# Excluir padrões:
rsync -avz --exclude='*.log' \
  --exclude='node_modules' /app/ user@server:/app/

# Deletar arquivos no destino que não existem na origem:
rsync -avz --delete /origem/ user@server:/destino/
# ⚠️ CUIDADO: pode deletar dados no destino!

# Dry-run (simular sem executar):
rsync -avzn --delete /origem/ destino/

# Limitar bandwidth:
rsync -avz --bwlimit=5000 origem/ destino/  # 5MB/s

# Porta SSH diferente:
rsync -avz -e "ssh -p 2222" /dados/ user@server:/backup/

# Backup incremental com hardlinks:
rsync -avz --link-dest=../yesterday \
  /dados/ /backups/today/
💡 rsync vs scp: Use scp para cópias simples e pontuais. Use rsync para sincronização, backups incrementais e transferências de grandes volumes — ele só transmite as diferenças, economizando tempo e bandwidth.

Criando um service unit fileCustom Service
# Arquivo: /etc/systemd/system/minha-app.service
[Unit]
Description=Minha Aplicação Node.js
Documentation=https://github.com/meu/app
After=network.target          # iniciar após rede estar pronta
Wants=postgresql.service      # dependência opcional
Requires=redis.service        # dependência obrigatória

[Service]
Type=simple                    # simple, forking, oneshot, notify
User=appuser                   # usuário para executar
Group=appgroup
WorkingDirectory=/opt/minha-app
Environment=NODE_ENV=production
Environment=PORT=3000
EnvironmentFile=/opt/minha-app/.env   # carregar variáveis de arquivo

ExecStartPre=/usr/bin/npm install     # executar antes de iniciar
ExecStart=/usr/bin/node server.js     # comando principal
ExecReload=/bin/kill -HUP $MAINPID    # comando de reload

# Restart automático em caso de falha:
Restart=on-failure
RestartSec=5                   # esperar 5s antes de reiniciar
StartLimitBurst=3              # máximo 3 restarts
StartLimitIntervalSec=60      # dentro de 60s

# Limites de recursos:
LimitNOFILE=65536
MemoryMax=512M
CPUQuota=50%

# Segurança:
NoNewPrivileges=yes
PrivateTmp=yes
ProtectSystem=strict
ReadWritePaths=/opt/minha-app/data

# Logging:
StandardOutput=journal
StandardError=journal
SyslogIdentifier=minha-app

[Install]
WantedBy=multi-user.target     # equivalente a "runlevel 3/5"

# ─── APÓS CRIAR O ARQUIVO: ───
sudo systemctl daemon-reload            # recarregar systemd
sudo systemctl enable --now minha-app   # habilitar e iniciar
sudo systemctl status minha-app         # verificar status
journalctl -u minha-app -f             # ver logs em tempo real
💡 Tipos de serviço:
  • simple — Processo principal roda no foreground (Node.js, Python)
  • forking — Processo faz fork e o pai termina (Apache, nginx com daemon mode)
  • oneshot — Executa uma vez e termina (scripts de inicialização)
  • notify — Envia notificação quando pronto (systemd-aware apps)

Cron

Cron Jobs — Agendamento de Tarefas

Cron é o agendador de tarefas do Linux. Permite executar scripts e comandos automaticamente em horários definidos — essencial para backups, limpeza de logs, monitoramento e automação.

Sintaxe do crontab

Formato da expressão cronSintaxe
# ┌───────────── minuto (0-59)
# │ ┌───────────── hora (0-23)
# │ │ ┌───────────── dia do mês (1-31)
# │ │ │ ┌───────────── mês (1-12 ou jan-dec)
# │ │ │ │ ┌───────────── dia da semana (0-7, 0 e 7 = domingo, ou sun-sat)
# │ │ │ │ │
# * * * * * comando_a_executar

# Caracteres especiais:
# *     = qualquer valor
# ,     = lista (1,3,5 = seg, qua, sex)
# -     = range (1-5 = segunda a sexta)
# /     = intervalo (*/5 = a cada 5)

# EXEMPLOS:
# ┌─ min  hora  dia  mês  sem  comando
  0  2    *     *    *    /opt/backup.sh          # Diário às 02:00
  */5 *   *     *    *    /opt/health-check.sh    # A cada 5 minutos
  0  */4  *     *    *    /opt/sync.sh            # A cada 4 horas
  30 8    *     *    1-5  /opt/relatorio.sh       # Seg-Sex às 08:30
  0  0    1     *    *    /opt/mensal.sh          # Dia 1 de cada mês à 00:00
  0  6    *     *    0    /opt/semanal.sh         # Todo domingo às 06:00
  0  0    1     1    *    /opt/anual.sh           # 1 de janeiro à 00:00
  */15 *  *     *    *    /opt/monitor.sh         # A cada 15 minutos
  0  9-18 *     *    1-5  /opt/horario-comercial.sh  # Seg-Sex, 9h-18h, a cada hora
💡 Atalhos do cron: 
@reboot    → Executar uma vez ao iniciar o sistema
@yearly    → 0 0 1 1 * (uma vez por ano, 1 jan 00:00)
@monthly   → 0 0 1 * * (uma vez por mês, dia 1 00:00)
@weekly    → 0 0 * * 0 (uma vez por semana, domingo 00:00)
@daily     → 0 0 * * * (uma vez por dia, 00:00)
@hourly    → 0 * * * * (uma vez por hora, minuto 0)

# Exemplo com atalho:
@daily /opt/backup.sh
@reboot /opt/start-app.sh

Exemplo prático: Cron para backup automático

Setup completo de backup agendadoExemplo
# 1. Criar o script de backup:
cat > /opt/scripts/backup-daily.sh << 'SCRIPT'
#!/bin/bash
set -euo pipefail

BACKUP_DIR="/opt/backups"
DATE=$(date +%Y%m%d-%H%M%S)
LOG="/var/log/backup-daily.log"
RETENTION=7

log() { echo "[$(date '+%F %T')] $1" >> "$LOG"; }

log "=== Backup diário iniciado ==="

# Backup de dados:
tar czf "$BACKUP_DIR/app-$DATE.tar.gz" /var/www/app/ 2>> "$LOG"

# Backup de banco:
pg_dump -U postgres meudb | gzip > "$BACKUP_DIR/db-$DATE.sql.gz" 2>> "$LOG"

# Limpar backups antigos:
find "$BACKUP_DIR" -name "*.gz" -mtime +$RETENTION -delete

log "=== Backup diário concluído ==="
SCRIPT

# 2. Tornar executável:
chmod +x /opt/scripts/backup-daily.sh

# 3. Testar manualmente:
/opt/scripts/backup-daily.sh

# 4. Adicionar ao crontab:
crontab -e
# Adicionar:
0 2 * * * /opt/scripts/backup-daily.sh >> /var/log/backup-daily.log 2>&1

# 5. Verificar se foi adicionado:
crontab -l

# 6. Monitorar execução:
tail -f /var/log/backup-daily.log

Exercício 2: Criar serviço systemd para aplicação

Crie um service unit file para uma aplicação Python (Flask) que roda na porta 5000 com gunicorn. O serviço deve: usar o usuário "webapp", reiniciar automaticamente em caso de falha, e iniciar após a rede estar pronta. Habilite e verifique o serviço.

▶ Ver solução 
# 1. Criar o unit file:
sudo cat > /etc/systemd/system/flask-app.service << 'EOF'
[Unit]
Description=Flask Application with Gunicorn
After=network.target

[Service]
Type=simple
User=webapp
Group=webapp
WorkingDirectory=/opt/flask-app
Environment=FLASK_ENV=production
ExecStart=/opt/flask-app/venv/bin/gunicorn -w 4 -b 0.0.0.0:5000 app:app
Restart=on-failure
RestartSec=5
StartLimitBurst=3
StartLimitIntervalSec=60
StandardOutput=journal
StandardError=journal
SyslogIdentifier=flask-app

[Install]
WantedBy=multi-user.target
EOF

# 2. Recarregar systemd:
sudo systemctl daemon-reload

# 3. Habilitar e iniciar:
sudo systemctl enable --now flask-app

# 4. Verificar status:
sudo systemctl status flask-app

# 5. Ver logs:
journalctl -u flask-app -f

Exercício 3: SSH seguro + cron de backup

Configure SSH com chave Ed25519 para um servidor remoto (sem senha), crie um alias no SSH config chamado "prod-server", e agende um rsync diário às 03:00 que sincroniza /var/www para o servidor remoto em /backups/web.

▶ Ver solução 
# 1. Gerar chave SSH:
ssh-keygen -t ed25519 -f ~/.ssh/prod-key -N "" -C "backup@meuservidor"

# 2. Copiar chave para o servidor:
ssh-copy-id -i ~/.ssh/prod-key.pub admin@54.100.200.50

# 3. Configurar SSH config (~/.ssh/config):
cat >> ~/.ssh/config << 'EOF'
Host prod-server
    HostName 54.100.200.50
    User admin
    Port 22
    IdentityFile ~/.ssh/prod-key
    ServerAliveInterval 60
EOF

# 4. Testar conexão:
ssh prod-server "hostname && uptime"

# 5. Criar script de backup:
cat > /opt/scripts/sync-web.sh << 'SCRIPT'
#!/bin/bash
set -euo pipefail
LOG="/var/log/sync-web.log"
echo "[$(date)] Iniciando sync..." >> "$LOG"
rsync -avz --delete /var/www/ prod-server:/backups/web/ >> "$LOG" 2>&1
echo "[$(date)] Sync concluído." >> "$LOG"
SCRIPT
chmod +x /opt/scripts/sync-web.sh

# 6. Agendar no cron (03:00 diariamente):
(crontab -l; echo "0 3 * * * /opt/scripts/sync-web.sh") | crontab -

# 7. Verificar:
crontab -l | grep sync-web
